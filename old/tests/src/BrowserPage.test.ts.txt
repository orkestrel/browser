/**
 * BrowserPage tests.
 *
 * Uses real Playwright browser instances — no mocks.
 * Covers Page targets, Frame targets, navigation, content extraction,
 * screenshots, element interaction, and edge cases.
 */

import type { Browser as PwBrowser, BrowserContext as PwContext } from 'playwright'
import { describe, it, expect, beforeAll, afterAll } from 'vitest'
import { chromium } from 'playwright'
import { BrowserPage, BrowserError } from 'keepalive'

// === Shared Playwright browser

let sharedBrowser: PwBrowser
let pwContext: PwContext

beforeAll(async () => {
	sharedBrowser = await chromium.launch({ headless: true })
	pwContext = await sharedBrowser.newContext()
}, 30_000)

afterAll(async () => {
	await pwContext.close()
	await sharedBrowser.close()
})

// === BrowserPage (Page target)

describe('BrowserPage', () => {
	describe('constructor', () => {
		it('wraps a real page target', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.url).toBe('about:blank')
			expect(bp.closed).toBe(false)
			await page.close()
		})

		it('throws BrowserError on null', () => {
			expect(() => new BrowserPage(null, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on undefined', () => {
			expect(() => new BrowserPage(undefined, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on empty object', () => {
			expect(() => new BrowserPage({}, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on string', () => {
			expect(() => new BrowserPage('string', false)).toThrow(BrowserError)
		})

		it('throws BrowserError on number', () => {
			expect(() => new BrowserPage(42, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on boolean', () => {
			expect(() => new BrowserPage(true, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on array', () => {
			expect(() => new BrowserPage([], false)).toThrow(BrowserError)
		})

		it('throws BrowserError on partial object missing methods', () => {
			expect(() => new BrowserPage({ url: () => '' }, false)).toThrow(BrowserError)
		})

		it('throws BrowserError on object with wrong method types', () => {
			expect(() => new BrowserPage({ url: 'not-a-function' }, false)).toThrow(BrowserError)
		})
	})

	describe('url', () => {
		it('returns about:blank for fresh page', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.url).toBe('about:blank')
			await page.close()
		})

		it('reflects navigation', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.url).toBe('about:blank')
			await bp.navigate('data:text/html,<h1>Nav</h1>')
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})

		it('returns data URI after setContent navigation', async () => {
			const page = await pwContext.newPage()
			await page.goto('data:text/html,<p>Direct</p>')
			const bp = new BrowserPage(page, false)
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})
	})

	describe('closed', () => {
		it('returns false for open page', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.closed).toBe(false)
			await page.close()
		})

		it('reflects page state after external close', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.closed).toBe(false)
			await page.close()
			expect(bp.closed).toBe(true)
		})
	})

	describe('title()', () => {
		it('returns the document title', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<html><head><title>Test Title</title></head><body></body></html>')
			const bp = new BrowserPage(page, false)
			expect(await bp.title()).toBe('Test Title')
			await page.close()
		})

		it('returns empty string for page without title', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<html><body>No Title</body></html>')
			const bp = new BrowserPage(page, false)
			expect(await bp.title()).toBe('')
			await page.close()
		})

		it('returns unicode title', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<html><head><title>日本語テスト</title></head></html>')
			const bp = new BrowserPage(page, false)
			expect(await bp.title()).toBe('日本語テスト')
			await page.close()
		})

		it('returns title with special characters', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<html><head><title>&lt;tag&gt; &amp; "quotes"</title></head></html>')
			const bp = new BrowserPage(page, false)
			expect(await bp.title()).toBe('<tag> & "quotes"')
			await page.close()
		})
	})

	describe('navigate()', () => {
		it('changes the page URL', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.navigate('data:text/html,<h1>Navigated</h1>')
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})

		it('respects condition option', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.navigate('data:text/html,<h1>Load</h1>', { condition: 'load' })
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})

		it('respects domcontentloaded condition', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.navigate('data:text/html,<h1>DOMContentLoaded</h1>', {
				condition: 'domcontentloaded',
			})
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})

		it('respects commit condition', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.navigate('data:text/html,<h1>Commit</h1>', { condition: 'commit' })
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})

		it('passes timeout option', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.navigate('data:text/html,<h1>Timeout</h1>', { timeout: 10_000 })
			expect(bp.url).toContain('data:text/html')
			await page.close()
		})
	})

	describe('content()', () => {
		it('returns url, title, html, and text', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<html><head><title>Content Test</title></head><body><p>Hello World</p></body></html>',
			)
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.title).toBe('Content Test')
			expect(result.html).toContain('<p>Hello World</p>')
			expect(result.text).toContain('Hello World')
			expect(result.url).toBeDefined()
			await page.close()
		})

		it('extracts visible text without markup', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<body><div>Visible <span>Text</span></div></body>')
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.text).toContain('Visible')
			expect(result.text).toContain('Text')
			expect(result.text).not.toContain('<div>')
			expect(result.text).not.toContain('<span>')
			await page.close()
		})

		it('returns empty text for empty body', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<html><body></body></html>')
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.text).toBe('')
			await page.close()
		})

		it('excludes hidden elements from text', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<body><div style="display:none">Hidden</div><div>Visible</div></body>')
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.text).toContain('Visible')
			expect(result.text).not.toContain('Hidden')
			await page.close()
		})

		it('captures html with nested elements', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<body><ul><li>One</li><li>Two</li><li>Three</li></ul></body>')
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.html).toContain('<ul>')
			expect(result.html).toContain('<li>One</li>')
			expect(result.html).toContain('<li>Three</li>')
			await page.close()
		})

		it('captures unicode content', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<html><head><title>Unicode</title></head><body>こんにちは世界 🌍</body></html>',
			)
			const bp = new BrowserPage(page, false)
			const result = await bp.content()
			expect(result.title).toBe('Unicode')
			expect(result.text).toContain('こんにちは世界')
			expect(result.text).toContain('🌍')
			await page.close()
		})
	})

	describe('screenshot()', () => {
		it('returns PNG bytes by default', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<h1>Screenshot</h1>')
			const bp = new BrowserPage(page, false)
			const result = await bp.screenshot()
			expect(result.bytes).toBeInstanceOf(Uint8Array)
			expect(result.bytes.length).toBeGreaterThan(0)
			// PNG magic bytes
			expect(result.bytes[0]).toBe(137)
			expect(result.bytes[1]).toBe(80)
			expect(result.bytes[2]).toBe(78)
			expect(result.bytes[3]).toBe(71)
			expect(result.path).toBeUndefined()
			await page.close()
		})

		it('returns JPEG bytes with jpeg type', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<h1>JPEG Test</h1>')
			const bp = new BrowserPage(page, false)
			const result = await bp.screenshot({ type: 'jpeg', quality: 80 })
			expect(result.bytes).toBeInstanceOf(Uint8Array)
			// JPEG magic bytes: FF D8
			expect(result.bytes[0]).toBe(0xff)
			expect(result.bytes[1]).toBe(0xd8)
			await page.close()
		})

		it('full page option captures more content', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div style="height:5000px">Tall Page</div>')
			const bp = new BrowserPage(page, false)
			const normal = await bp.screenshot()
			const full = await bp.screenshot({ full: true })
			expect(full.bytes.length).toBeGreaterThan(normal.bytes.length)
			await page.close()
		})

		it('returns non-empty bytes for blank page', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.screenshot()
			expect(result.bytes).toBeInstanceOf(Uint8Array)
			expect(result.bytes.length).toBeGreaterThan(0)
			await page.close()
		})

		it('captures colored content', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<body style="background:red"><h1 style="color:white">Red Page</h1></body>',
			)
			const bp = new BrowserPage(page, false)
			const result = await bp.screenshot()
			// Just verify it's a valid PNG with content
			expect(result.bytes[0]).toBe(137)
			expect(result.bytes.length).toBeGreaterThan(100)
			await page.close()
		})
	})

	describe('click()', () => {
		it('clicks a real element', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<button id="btn" onclick="document.title=\'clicked\'">Click Me</button>',
			)
			const bp = new BrowserPage(page, false)
			await bp.click('#btn')
			expect(await bp.title()).toBe('clicked')
			await page.close()
		})

		it('clicks an element with timeout option', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<button id="btn2" onclick="document.title=\'ok\'">OK</button>')
			const bp = new BrowserPage(page, false)
			await bp.click('#btn2', { timeout: 5000 })
			expect(await bp.title()).toBe('ok')
			await page.close()
		})

		it('throws when selector does not exist', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div>No button</div>')
			const bp = new BrowserPage(page, false)
			await expect(bp.click('#nonexistent', { timeout: 500 })).rejects.toThrow()
			await page.close()
		})
	})

	describe('fill()', () => {
		it('types into a real input', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<input id="input" type="text" />')
			const bp = new BrowserPage(page, false)
			await bp.fill('#input', 'hello world')
			const value = await bp.evaluate('document.getElementById("input").value')
			expect(value).toBe('hello world')
			await page.close()
		})

		it('clears and fills an input that already has text', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<input id="input" type="text" value="old" />')
			const bp = new BrowserPage(page, false)
			await bp.fill('#input', 'new')
			const value = await bp.evaluate('document.getElementById("input").value')
			expect(value).toBe('new')
			await page.close()
		})

		it('fills a textarea', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<textarea id="ta"></textarea>')
			const bp = new BrowserPage(page, false)
			await bp.fill('#ta', 'multi\nline\ntext')
			const value = await bp.evaluate('document.getElementById("ta").value')
			expect(value).toBe('multi\nline\ntext')
			await page.close()
		})

		it('fills with unicode characters', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<input id="input" type="text" />')
			const bp = new BrowserPage(page, false)
			await bp.fill('#input', '日本語 🎉')
			const value = await bp.evaluate('document.getElementById("input").value')
			expect(value).toBe('日本語 🎉')
			await page.close()
		})

		it('fills with empty string to clear', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<input id="input" type="text" value="filled" />')
			const bp = new BrowserPage(page, false)
			await bp.fill('#input', '')
			const value = await bp.evaluate('document.getElementById("input").value')
			expect(value).toBe('')
			await page.close()
		})
	})

	describe('select()', () => {
		it('selects an option in a dropdown', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<select id="sel"><option value="a">A</option><option value="b">B</option><option value="c">C</option></select>',
			)
			const bp = new BrowserPage(page, false)
			await bp.select('#sel', ['b'])
			const value = await bp.evaluate('document.getElementById("sel").value')
			expect(value).toBe('b')
			await page.close()
		})

		it('selects the first option', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<select id="sel"><option value="x">X</option><option value="y">Y</option></select>',
			)
			const bp = new BrowserPage(page, false)
			await bp.select('#sel', ['x'])
			const value = await bp.evaluate('document.getElementById("sel").value')
			expect(value).toBe('x')
			await page.close()
		})

		it('selects the last option', async () => {
			const page = await pwContext.newPage()
			await page.setContent(
				'<select id="sel"><option value="1">1</option><option value="2">2</option><option value="3">3</option></select>',
			)
			const bp = new BrowserPage(page, false)
			await bp.select('#sel', ['3'])
			const value = await bp.evaluate('document.getElementById("sel").value')
			expect(value).toBe('3')
			await page.close()
		})
	})

	describe('evaluate()', () => {
		it('executes JavaScript in the page', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('1 + 2')
			expect(result).toBe(3)
			await page.close()
		})

		it('can access the DOM', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div id="target">Hello</div>')
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('document.getElementById("target").textContent')
			expect(result).toBe('Hello')
			await page.close()
		})

		it('returns null for null expression result', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('null')
			expect(result).toBeNull()
			await page.close()
		})

		it('returns undefined for undefined expression result', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('undefined')
			expect(result).toBeUndefined()
			await page.close()
		})

		it('returns boolean values', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(await bp.evaluate('true')).toBe(true)
			expect(await bp.evaluate('false')).toBe(false)
			await page.close()
		})

		it('returns string values', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('"hello world"')
			expect(result).toBe('hello world')
			await page.close()
		})

		it('returns objects', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('({a: 1, b: "two"})')
			expect(result).toEqual({ a: 1, b: 'two' })
			await page.close()
		})

		it('returns arrays', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('[1, 2, 3]')
			expect(result).toEqual([1, 2, 3])
			await page.close()
		})

		it('handles math operations', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(await bp.evaluate('Math.PI')).toBeCloseTo(3.14159, 4)
			expect(await bp.evaluate('Math.max(1, 5, 3)')).toBe(5)
			await page.close()
		})

		it('reads window properties', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const result = await bp.evaluate('typeof window')
			expect(result).toBe('object')
			await page.close()
		})
	})

	describe('wait()', () => {
		it('resolves for a selector that exists', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div id="ready">Present</div>')
			const bp = new BrowserPage(page, false)
			await expect(bp.wait('#ready', { timeout: 5000 })).resolves.toBeUndefined()
			await page.close()
		})

		it('resolves for dynamically added element', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div id="container"></div>')
			const bp = new BrowserPage(page, false)
			// Add visible element after a short delay
			await page.evaluate(() => {
				setTimeout(() => {
					const el = document.createElement('div')
					el.id = 'dynamic'
					el.textContent = 'Loaded'
					el.style.display = 'block'
					document.getElementById('container')?.appendChild(el)
				}, 50)
			})
			await expect(bp.wait('#dynamic', { timeout: 5000 })).resolves.toBeUndefined()
			await page.close()
		}, 10_000)

		it('rejects when selector does not appear in time', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<div>No target</div>')
			const bp = new BrowserPage(page, false)
			await expect(bp.wait('#missing', { timeout: 200 })).rejects.toThrow()
			await page.close()
		})
	})

	describe('frame()', () => {
		it('finds a named iframe', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<iframe name="child" srcdoc="<h1>Inside Frame</h1>"></iframe>')
			await page.waitForTimeout(300)
			const bp = new BrowserPage(page, false)
			const child = bp.frame('child')
			expect(child).toBeDefined()
			await page.close()
		})

		it('returns undefined for nonexistent frame', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.frame('nonexistent')).toBeUndefined()
			await page.close()
		})

		it('returned frame has correct url', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<iframe name="child" src="about:blank"></iframe>')
			await page.waitForTimeout(300)
			const bp = new BrowserPage(page, false)
			const child = bp.frame('child')
			expect(child).toBeDefined()
			expect(child?.url).toBe('about:blank')
			await page.close()
		})
	})

	describe('frames()', () => {
		it('returns empty for a Page target without childFrames', async () => {
			const page = await pwContext.newPage()
			await page.setContent('<iframe name="f1" src="about:blank"></iframe>')
			await page.waitForTimeout(300)
			const bp = new BrowserPage(page, false)
			// Page does not have childFrames(), so frames() returns empty
			expect(bp.frames()).toHaveLength(0)
			await page.close()
		})

		it('returns immutable array', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			const frames = bp.frames()
			expect(Array.isArray(frames)).toBe(true)
			await page.close()
		})
	})

	describe('close()', () => {
		it('closes the page', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			expect(bp.closed).toBe(false)
			await bp.close()
			expect(bp.closed).toBe(true)
		})

		it('double close is safe', async () => {
			const page = await pwContext.newPage()
			const bp = new BrowserPage(page, false)
			await bp.close()
			// Second close should not throw — page.close() on an already closed page
			// may throw from Playwright, but the isClosed check protects
			expect(bp.closed).toBe(true)
		})
	})
})

// === BrowserPage (Frame target)

describe('BrowserPage (Frame target)', () => {
	it('wraps a real Frame target', async () => {
		const page = await pwContext.newPage()
		await page.setContent('<iframe name="child" src="about:blank"></iframe>')
		await page.waitForTimeout(300)
		const frame = page.mainFrame().childFrames()[0]
		const bp = new BrowserPage(frame, true)
		expect(bp.url).toBe('about:blank')
		expect(bp.closed).toBe(false)
		await page.close()
	})

	it('detects detached state via isDetached', async () => {
		const page = await pwContext.newPage()
		await page.setContent('<iframe name="child" src="about:blank"></iframe>')
		await page.waitForTimeout(300)
		const frame = page.mainFrame().childFrames()[0]
		const bp = new BrowserPage(frame, true)
		expect(bp.closed).toBe(false)
		// Remove the iframe to detach the frame
		await page.evaluate('document.querySelector("iframe").remove()')
		await page.waitForTimeout(100)
		expect(bp.closed).toBe(true)
		await page.close()
	})
