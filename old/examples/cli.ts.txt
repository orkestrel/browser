// === Reactive State
//
// All mutable state lives in signals. The queue is imperative, so its
// properties are mirrored into signals by a lightweight polling loop.
// The dashboard is a single Compute that reads every signal — when
// any dependency changes, the Watch fires and repaints the terminal.

const scope = createScope()

// Configuration
const concurrency = createSignal(initialConcurrency)
const timeout = createSignal(initialTimeout)
const retries = createSignal(initialRetries)

// Lifecycle
const destroyed = createSignal(false)
const shuttingDown = createSignal(false)
const taskCounter = createSignal(0)

// Worker state mirrors
const workerCompleted = createSignal(0)
const workerFailed = createSignal(0)
const workerActive = createSignal(0)
const workerPoolSize = createSignal(0)
const workerDestroyed = createSignal(false)

// NodeWorker state mirrors
const nodeWorkerCompleted = createSignal(0)
const nodeWorkerFailed = createSignal(0)
const nodeWorkerActive = createSignal(0)
const nodeWorkerThreads = createSignal(0)
const nodeWorkerDestroyed = createSignal(false)

// Queue state mirrors
const queueCount = createSignal(0)
const queueActive = createSignal(0)
const queueCompleted = createSignal(0)
const queueFailed = createSignal(0)
const queuePaused = createSignal(false)
const queueStopped = createSignal(false)
const queueAborted = createSignal(false)
const queueBailed = createSignal(false)
const queueIdle = createSignal(true)

function entriesEqual(
	a: readonly QueueEntryState<string>[],
	b: readonly QueueEntryState<string>[],
): boolean {
	if (a.length !== b.length) return false
	for (let i = 0; i < a.length; i++) {
		if (a[i].id !== b[i].id || a[i].status !== b[i].status || a[i].attempts !== b[i].attempts)
			return false
	}
	return true
}

const queueTracked = createSignal<readonly QueueEntryState<string>[]>([], {
	equal: entriesEqual,
})

// === Activity Log
//
// Messages are pushed through a Channel. A single subscriber formats
// them with a timestamp and feeds the logEntries signal. The Compute
// that builds the screen reads logEntries, so any new message
// automatically triggers a repaint — no manual requestRender needed.

const MAX_LOG_ENTRIES = 14
const logChannel = createChannel<string>()
const logEntries = createSignal<readonly string[]>([])

const unsubscribeLog = logChannel.subscribe((message) => {
	const timestamp = new Date().toLocaleTimeString('en-US', { hour12: false })
	const formatted = `${DIM}${timestamp}${RESET} ${message}`
	logEntries.update((current) => {
		const next = [...current, formatted]
		return next.length > MAX_LOG_ENTRIES ? next.slice(next.length - MAX_LOG_ENTRIES) : next
	})
})

scope.add(unsubscribeLog, logChannel)

function log(message: string): void {
	logChannel.send(message)
}

// === Queue Setup

const store = createJsonQueueStore<string>(storePath)
const workerStore = createJsonQueueStore<string>(workerStorePath)
const nodeWorkerStore = createJsonQueueStore<number>(nodeWorkerStorePath)

const queue = createQueue<string, string>({
	concurrency: initialConcurrency,
	timeout: initialTimeout,
	retries: initialRetries,
	store,
	handler: async (taskName) => {
		const start = Date.now()
		const handlerTimeout = timeout.peek()

		// Simulate variable workloads: normal, slow (timeout), or buggy (failure)
		const typeRoll = Math.random()
		let duration = 400 + Math.floor(Math.random() * 1200)
		let shouldFail = false

		if (typeRoll > 0.88) {
			duration = handlerTimeout + 500
		} else if (typeRoll > 0.68) {
			shouldFail = true
		}

		while (Date.now() - start < duration) {
			if (queue.aborted || queue.stopped) {
				throw new Error('Task cancelled mid-flight')
			}
			await new Promise((resolve) => setTimeout(resolve, 40))
		}

		if (shouldFail) {
			throw new Error(`Random failure on ${taskName}`)
		}

		return taskName
	},
})

queue.start()

// === Worker Setup
//
// Resource-backed worker using createWorker. The pooled "resource"
// is a simple counter object simulating a reusable connection.

let resourceCounter = 0

interface SimulatedResource {
	readonly id: number
	readonly label: string
}

const worker = createWorker<string, SimulatedResource, string>({
	handler: async (taskName, resource) => {
		const duration = 300 + Math.floor(Math.random() * 700)
		await new Promise((resolve) => setTimeout(resolve, duration))

		const typeRoll = Math.random()
		if (typeRoll > 0.82) {
			throw new Error(`Worker failure on ${taskName}`)
		}

		return `${taskName} [res:${resource.label}]`
	},
	pool: {
		create: async () => {
			resourceCounter++
			return { id: resourceCounter, label: `R${resourceCounter}` }
		},
		destroy: async () => {},
	},
	concurrency: 2,
	timeout: 3000,
	retries: 1,
	store: workerStore,
})

worker.start()

// === Terminal Entry Watcher
//
// Scans queue entries for terminal statuses (completed, failed, aborted,
// expired) and logs them via the channel. Called inside each poll batch
// so new results and state updates coalesce into a single render pass.

const loggedResults = new Set<string>()

function isTerminalStatus(status: string): boolean {
	return (
		status === 'completed' || status === 'failed' || status === 'aborted' || status === 'expired'
	)
}

function entryTagline(entry: QueueEntryState<string>): string {
	const tags: string[] = []
	if (entry.options?.sequential) tags.push('seq')
	if (entry.options?.bail) tags.push('bail')
	if (entry.options?.priority !== undefined && entry.options.priority > 0)
		tags.push(`p:${entry.options.priority}`)
	if (entry.attempts > 1) tags.push(`×${entry.attempts}`)
	return tags.length > 0 ? ` ${DIM}(${tags.join(', ')})${RESET}` : ''
}

function checkTerminalEntries(): void {
	for (const entry of queue.entries()) {
		if (loggedResults.has(entry.id)) continue
		if (!isTerminalStatus(entry.status)) continue

		loggedResults.add(entry.id)

		const name = String(entry.context)
		const flags = entryTagline(entry)

		switch (entry.status) {
			case 'completed':
				log(`${GREEN}✓${RESET} ${BOLD}${name}${RESET} completed${flags}`)
				break
			case 'failed':
				log(`${RED}✗${RESET} ${BOLD}${name}${RESET} failed${flags}`)
				break
			case 'aborted':
				log(`${RED}⊘${RESET} ${BOLD}${name}${RESET} aborted${flags}`)
				break
			case 'expired':
				log(`${YELLOW}⏱${RESET} ${BOLD}${name}${RESET} expired${flags}`)
				break
		}
	}
}

// === Queue State Sync
//
// The queue is not reactive — its getters are plain reads. These helpers
// mirror the queue's current state into signals. Wrapping the update in
// batcher.run() coalesces all signal changes into one flush, so the
// screen compute recomputes at most once per sync cycle.

function syncImmediate(): void {
	queueCount.set(queue.count)
	queueActive.set(queue.active)
	queueCompleted.set(queue.completed)
	queueFailed.set(queue.failed)
	queuePaused.set(queue.paused)
	queueStopped.set(queue.stopped)
	queueAborted.set(queue.aborted)
	queueBailed.set(queue.bailed)
	queueIdle.set(queue.idle)
	queueTracked.set(queue.entries())

	// Worker state
	workerCompleted.set(worker.completed)
	workerFailed.set(worker.failed)
	workerActive.set(worker.active)
	workerPoolSize.set(worker.stats.size)
	workerDestroyed.set(worker.destroyed)

	// NodeWorker state
	nodeWorkerCompleted.set(nodeWorkerDouble.completed + nodeWorkerSlow.completed)
	nodeWorkerFailed.set(nodeWorkerDouble.failed + nodeWorkerSlow.failed)
	nodeWorkerActive.set(nodeWorkerDouble.active + nodeWorkerSlow.active)
	nodeWorkerThreads.set(nodeWorkerDouble.stats.threads + nodeWorkerSlow.stats.threads)
	nodeWorkerDestroyed.set(nodeWorkerDouble.destroyed && nodeWorkerSlow.destroyed)
}

function syncQueueState(): void {
	batcher.run(() => {
		checkTerminalEntries()
		syncImmediate()
	})
}

// === Polling
//
// Adaptive timer that reads queue state into signals. Backs off
// exponentially when nothing changes to minimize CPU cost during
// idle periods, and polls aggressively when workers are active.

const INITIAL_POLL_MS = 40
const ACTIVE_POLL_MS = 70
const PENDING_POLL_MS = 100
const MAX_POLL_MS = 280

let pollTimer: ReturnType<typeof setTimeout> | undefined
let unchangedPollCount = 0

function shouldPoll(): boolean {
	if (shuttingDown.peek()) return false
	if (queue.active > 0) return true
	if (queue.idle) return false
	return !queue.paused && !queue.stopped && !queue.aborted && !destroyed.peek()
}

function pollDelay(): number {
	if (unchangedPollCount === 0) return INITIAL_POLL_MS
	const base = queue.active > 0 ? ACTIVE_POLL_MS : PENDING_POLL_MS
	const multiplier = Math.pow(2, Math.min(unchangedPollCount - 1, 2))
	return Math.min(base * multiplier, MAX_POLL_MS)
}

function stopPolling(): void {
	if (pollTimer === undefined) return
	clearTimeout(pollTimer)
	pollTimer = undefined
}

function schedulePoll(): void {
	if (!shouldPoll() || pollTimer !== undefined) return

	pollTimer = setTimeout(() => {
		pollTimer = undefined

		const beforeEntries = queueTracked.peek()
		syncQueueState()

		if (queueTracked.peek() !== beforeEntries) {
			unchangedPollCount = 0
		} else {
			unchangedPollCount++
		}

		if (shouldPoll()) schedulePoll()
	}, pollDelay())
}

function startPolling(): void {
	unchangedPollCount = 0
	stopPolling()
	schedulePoll()
}

// === Dashboard Formatting
//
// Each formatter reads signals via .get(), which registers dependencies
// on the screen Compute. Changes to any signal mark the Compute dirty,
// and the Watch lazily recomputes and repaints only when the final
// screen string actually differs.

function formatStateLabel(): string {
	if (destroyed.get()) return `${BG_RED}${WHITE}${BOLD} DESTROYED ${RESET}`
	if (queueAborted.get()) return `${BG_RED}${WHITE}${BOLD} ABORTED ${RESET}`
	if (queueStopped.get()) return `${BG_YELLOW}${WHITE}${BOLD} STOPPED ${RESET}`
	if (queueBailed.get()) return `${BG_YELLOW}${WHITE}${BOLD} BAILED ${RESET}`
	if (queuePaused.get()) return `${BG_YELLOW}${WHITE}${BOLD} PAUSED ${RESET}`
	return `${BG_GREEN}${WHITE}${BOLD} RUNNING ${RESET}`
}

function formatWorkerBar(): string {
	const active = queueActive.get()
	const limit = concurrency.get()
	const filled = Math.min(active, limit)
	const empty = limit - filled
	const bar = `${GREEN}${'█'.repeat(filled)}${DIM}${'░'.repeat(empty)}${RESET}`
	return `[${bar}] ${active}/${limit}`
}

function formatSuccessRate(): string {
	const completed = queueCompleted.get()
	const failed = queueFailed.get()
	const total = completed + failed
	if (total === 0) return `${DIM}—${RESET}`
	const rate = Math.round((completed / total) * 100)
	const color = rate >= 80 ? GREEN : rate >= 50 ? YELLOW : RED
	return `${color}${rate}%${RESET}`
}

function statusBadge(status: string): string {
	switch (status) {
		case 'pending':
			return `${YELLOW}●${RESET}`
		case 'scheduled':
			return `${MAGENTA}◈${RESET}`
		case 'active':
			return `${CYAN}◉${RESET}`
		case 'completed':
			return `${GREEN}✓${RESET}`
		case 'failed':
			return `${RED}✗${RESET}`
		case 'aborted':
			return `${RED}⊘${RESET}`
		case 'expired':
			return `${DIM}⊘${RESET}`
		default:
			return `${DIM}?${RESET}`
	}
}

function entryFlags(entry: QueueEntryState<string>): string {
	const flags: string[] = []
	if (entry.options?.sequential) flags.push(`${MAGENTA}seq${RESET}`)
	if (entry.options?.bail) flags.push(`${RED}bail${RESET}`)
	if (entry.options?.priority !== undefined && entry.options.priority > 0)
		flags.push(`${CYAN}p:${entry.options.priority}${RESET}`)
	if (entry.options?.retries !== undefined) flags.push(`${CYAN}r:${entry.options.retries}${RESET}`)
	if (entry.options?.timeout !== undefined)
		flags.push(`${YELLOW}t:${entry.options.timeout}${RESET}`)
	if (entry.options?.signal) flags.push(`${BLUE}sig${RESET}`)
	if (entry.options?.schedule?.delay !== undefined)
		flags.push(`${MAGENTA}d:${entry.options.schedule.delay}${RESET}`)
	if (entry.schedule?.activation !== undefined)
		flags.push(
			`${MAGENTA}@${new Date(entry.schedule.activation).toLocaleTimeString('en-US', { hour12: false })}${RESET}`,
		)
	if (entry.schedule?.expiration !== undefined)
		flags.push(
			`${DIM}exp:${new Date(entry.schedule.expiration).toLocaleTimeString('en-US', { hour12: false })}${RESET}`,
		)
	return flags.length > 0 ? ` ${DIM}[${RESET}${flags.join(DIM + ',' + RESET)}${DIM}]${RESET}` : ''
}

function formatEntryTable(): string {
	const all = queueTracked.get()
	if (all.length === 0) return `    ${DIM}No tracked entries${RESET}`

	const visible = all.slice(-10)
	const hidden = all.length - visible.length
	const lines: string[] = []

	if (hidden > 0) {
		lines.push(`    ${DIM}... ${hidden} earlier entries${RESET}`)
	}

	for (const entry of visible) {
		const badge = statusBadge(entry.status)
		const id = entry.id.length > 14 ? entry.id.slice(0, 14) + '…' : entry.id.padEnd(15)
		const ctx =
			String(entry.context).length > 12
				? String(entry.context).slice(0, 12) + '…'
				: String(entry.context).padEnd(13)
		const att = entry.attempts > 0 ? `${DIM}×${entry.attempts}${RESET}` : `${DIM}  ${RESET}`
		const flags = entryFlags(entry)
		lines.push(`    ${badge} ${DIM}${id}${RESET} ${ctx} ${att}${flags}`)
	}

	return lines.join('\n')
}

function formatActivityLog(): string {
	const entries = logEntries.get()
	if (entries.length === 0) return `    ${DIM}No activity yet${RESET}`
	return entries.map((entry) => `  ${entry}`).join('\n')
}

function formatWorkerStats(): string {
	const completed = workerCompleted.get()
	const failed = workerFailed.get()
	const active = workerActive.get()
	const pool = workerPoolSize.get()
	const isDestroyed = workerDestroyed.get()
	if (isDestroyed) return `${DIM}destroyed${RESET}`
	return `${GREEN}✓${completed}${RESET} ${RED}✗${failed}${RESET} ${CYAN}◉${active}${RESET} ${DIM}pool:${pool}${RESET}`
}

function formatNodeWorkerStats(): string {
	const completed = nodeWorkerCompleted.get()
	const failed = nodeWorkerFailed.get()
	const active = nodeWorkerActive.get()
	const threads = nodeWorkerThreads.get()
	const isDestroyed = nodeWorkerDestroyed.get()
	if (isDestroyed) return `${DIM}destroyed${RESET}`
	return `${GREEN}✓${completed}${RESET} ${RED}✗${failed}${RESET} ${CYAN}◉${active}${RESET} ${DIM}threads:${threads}${RESET}`
}

// === Task Enqueue Helpers

function guardEnqueue(): boolean {
	return !destroyed.peek() && !queue.aborted && !queue.stopped
}

function nextTaskId(prefix: string): { readonly name: string; readonly id: string } {
	taskCounter.update((n) => n + 1)
	const count = taskCounter.peek()
	return { name: `${prefix}-${count}`, id: `${prefix.toLowerCase()}-${count}` }
}

function enqueueTask(): void {
	if (!guardEnqueue()) return
	batcher.run(() => {
		const { name, id } = nextTaskId('Task')
		log(`${BLUE}+${RESET} Enqueued ${BOLD}${name}${RESET}`)
		queue.enqueue(name, { id }).catch(() => {})
		syncImmediate()
	})
	startPolling()
}

function enqueueSequential(): void {
	if (!guardEnqueue()) return
	batcher.run(() => {
		const { name, id } = nextTaskId('Seq')
		log(`${MAGENTA}⧫${RESET} Enqueued ${BOLD}${name}${RESET} ${DIM}(sequential)${RESET}`)
		queue.enqueue(name, { id, sequential: true }).catch(() => {})
		syncImmediate()
	})
	startPolling()
}

function enqueueBail(): void {
	if (!guardEnqueue()) return
	batcher.run(() => {
		const { name, id } = nextTaskId('Bail')
		log(`${RED}!${RESET} Enqueued ${BOLD}${name}${RESET} ${DIM}(bail on fail)${RESET}`)
		queue.enqueue(name, { id, bail: true }).catch(() => {})
		syncImmediate()
	})
	startPolling()
}

function enqueueWithSignal(): void {
	if (!guardEnqueue()) return
	const controller = new AbortController()

	batcher.run(() => {
		const { name, id } = nextTaskId('Sig')

		const cancelTimer = setTimeout(() => {
			controller.abort()
			batcher.run(() => log(`${YELLOW}⚡${RESET} Signal fired for ${BOLD}${name}${RESET}`))
		}, 1000)

		log(`${BLUE}⚡${RESET} Enqueued ${BOLD}${name}${RESET} ${DIM}(signal — auto-cancel 1s)${RESET}`)
		queue
			.enqueue(name, { id, signal: controller.signal })
			.then(() => clearTimeout(cancelTimer))
			.catch(() => clearTimeout(cancelTimer))
		syncImmediate()
	})

	startPolling()
}

function enqueueCustomTimeout(): void {
	if (!guardEnqueue()) return
	const shortTimeout = 500
	batcher.run(() => {
		const { name, id } = nextTaskId('Fast')
		log(
			`${CYAN}⏱${RESET} Enqueued ${BOLD}${name}${RESET} ${DIM}(timeout ${shortTimeout}ms)${RESET}`,
		)
		queue.enqueue(name, { id, timeout: shortTimeout, retries: 0 }).catch(() => {})
		syncImmediate()
	})
	startPolling()
}

function enqueueDelayed(): void {
	if (!guardEnqueue()) return
	const delayMs = 3000
	batcher.run(() => {
		const { name, id } = nextTaskId('Delay')

// === Shutdown

async function shutdown(): Promise<void> {
	shuttingDown.set(true)
	stopPolling()

	if (process.stdin.isTTY) {
		process.stdin.setRawMode(false)
	}
	process.stdin.pause()

	// Tear down the reactive graph
	scope.destroy()

	process.stdout.write('\x1Bc')
	console.log(`\n  ${BOLD}Shutting down...${RESET}`)

	// Force exit after 5 seconds if graceful shutdown stalls
	const forceTimer = setTimeout(() => {
		console.log(`  ${YELLOW}⚠${RESET} Forced exit after timeout`)
		process.exit(1)
	}, 5000)

	if (!destroyed.peek() && !queue.aborted) {
		await queue.shutdown()
		console.log(`  ${GREEN}✓${RESET} Queue state synced to ${DIM}${storePath}${RESET}`)
	}

	// Tear down Worker and NodeWorker instances
	if (!worker.destroyed) {
		await worker.shutdown()
		console.log(`  ${GREEN}✓${RESET} Worker state synced to ${DIM}${workerStorePath}${RESET}`)
	}
	if (!nodeWorkerDouble.destroyed) {
		await nodeWorkerDouble.destroy()
		console.log(`  ${GREEN}✓${RESET} NodeWorker (double) destroyed`)
	}
	if (!nodeWorkerSlow.destroyed) {
		await nodeWorkerSlow.shutdown()
		console.log(
			`  ${GREEN}✓${RESET} NodeWorker (slow) state synced to ${DIM}${nodeWorkerStorePath}${RESET}`,
		)
	}

	clearTimeout(forceTimer)

	const tracked = queue.entries()
	const completed = tracked.filter((e) => e.status === 'completed').length
	const failed = tracked.filter((e) => e.status === 'failed').length
	const aborted = tracked.filter((e) => e.status === 'aborted').length

	console.log(`  ${GREEN}✓${RESET} All workers shut down`)
	console.log(
		`  ${DIM}Queue — Tracked: ${tracked.length}  Completed: ${completed}  Failed: ${failed}  Aborted: ${aborted}${RESET}`,
	)
	console.log(`  ${DIM}Worker — Completed: ${worker.completed}  Failed: ${worker.failed}${RESET}`)
	console.log(
		`  ${DIM}NodeWorker — Completed: ${nodeWorkerDouble.completed + nodeWorkerSlow.completed}  Failed: ${nodeWorkerDouble.failed + nodeWorkerSlow.failed}${RESET}\n`,
	)

	process.exit(0)
}

// === Input Handling

if (process.stdin.isTTY) {
	process.stdin.setRawMode(true)
}
readline.emitKeypressEvents(process.stdin)
process.stdin.resume()

process.stdin.on('keypress', (str: string | undefined, key: Key) => {
	const keyName = key.name?.toLowerCase() ?? str?.toLowerCase()

	if ((key.ctrl === true && key.name === 'c') || keyName === 'x') {
		void shutdown()
		return
	}

	switch (keyName) {
		// --- Tasks ---
		case 'a':
			enqueueTask()
			break
		case 'b':
			for (let i = 0; i < 5; i++) enqueueTask()
			break
		case 'q':
			enqueueSequential()
			break
		case 'f':
			enqueueBail()
			break
		case 'e':
			enqueueWithSignal()
			break
		case 't':
			enqueueCustomTimeout()
			break
		case 'n':
			enqueueDelayed()
			break
		case 'i':
			enqueuePriority()
			break
		case 'm':
			enqueuePrepend()
			break
		case 'v':
			enqueueExpiring()
			break

		// --- Worker / NodeWorker ---
		case '1':
			enqueueWorkerTask()
			break
		case '2':
			for (let i = 0; i < 3; i++) enqueueWorkerTask()
			break
		case '3':
			enqueueNodeWorkerDouble()
			break
		case '4':
			for (let i = 0; i < 3; i++) enqueueNodeWorkerSlow()
			break
		case '5':
			enqueueNodeWorkerSlow()
			break

		// --- Lifecycle ---
		case 'g':
			if (destroyed.peek()) {
				batcher.run(() => log(`${RED}Cannot start — queue is destroyed${RESET}`))
			} else if (queue.aborted) {
				batcher.run(() => log(`${RED}Cannot start — queue is aborted${RESET}`))
			} else {
				queue.start()
				batcher.run(() => {
					log(`${GREEN}▶${RESET} Workers started`)
					syncImmediate()
				})
				startPolling()
			}
			break
		case 'p':
			if (destroyed.peek() || queue.aborted) {
				const reason = destroyed.peek() ? 'destroyed' : 'aborted'
				batcher.run(() => log(`${RED}Cannot pause — queue is ${reason}${RESET}`))
			} else {
				queue.pause()
				batcher.run(() => {
					log(`${YELLOW}⏸${RESET} Queue paused`)
					syncImmediate()
				})
			}
			break
		case 'r':
			if (destroyed.peek() || queue.aborted) {
				const reason = destroyed.peek() ? 'destroyed' : 'aborted'
				batcher.run(() => log(`${RED}Cannot resume — queue is ${reason}${RESET}`))
			} else if (queue.stopped) {
				queue.start()
				batcher.run(() => {
					log(`${GREEN}▶${RESET} Queue restarted`)
					syncImmediate()
				})
				startPolling()
			} else {
				queue.resume()
				batcher.run(() => {
					log(`${GREEN}▶${RESET} Queue resumed`)
					syncImmediate()
				})
				startPolling()
			}
			break
		case 'c':
			queue.clear()
			batcher.run(() => {
				log(`${YELLOW}⌀${RESET} Pending contexts cleared`)
				syncImmediate()
			})
			break

		// --- Configuration ---
		case 'up':
		case 'right':
			batcher.run(() => {
				concurrency.update((n) => n + 1)
				queue.configure({ concurrency: concurrency.peek() })
				log(`${CYAN}↑${RESET} Concurrency → ${BOLD}${concurrency.peek()}${RESET}`)
				syncImmediate()
			})
			startPolling()
			break
		case 'down':
		case 'left':
			if (concurrency.peek() > 1) {
				batcher.run(() => {
					concurrency.update((n) => n - 1)
					queue.configure({ concurrency: concurrency.peek() })
					log(`${CYAN}↓${RESET} Concurrency → ${BOLD}${concurrency.peek()}${RESET}`)
					syncImmediate()
				})
				startPolling()
			}
			break

		// --- Persistence ---
		case 'w':
			void Promise.all([queue.sync(), worker.sync(), nodeWorkerSlow.sync()]).then(() => {
				batcher.run(() => log(`${GREEN}💾${RESET} State synced to ${DIM}${storePath}${RESET}`))
			})
			break
		case 'l':
			void Promise.all([queue.restore(), worker.restore(), nodeWorkerSlow.restore()]).then(() => {
				for (const entry of queue.entries()) {
					if (isTerminalStatus(entry.status)) {
						loggedResults.add(entry.id)
					}
				}
				const restored = queue
					.entries()
					.filter((e) => e.status === 'pending' || e.status === 'scheduled').length
				batcher.run(() => {
					log(`${GREEN}📂${RESET} Restored ${BOLD}${restored}${RESET} entries from disk`)
					syncImmediate()
				})
				if (shouldPoll()) startPolling()
			})
			break

		// --- Stop / Abort / Destroy ---
		case 's':
			queue.stop()
			batcher.run(() => {
				log(`${YELLOW}■${RESET} Queue stopped`)
				syncImmediate()
			})
			stopPolling()
			break
		case 'k':
			queue.abort()
			batcher.run(() => {
				log(`${RED}✕${RESET} Queue aborted`)
				syncImmediate()
			})
			stopPolling()
			break
		case 'd':
			if (!destroyed.peek()) {
				queue.destroy()
				batcher.run(() => {
					destroyed.set(true)
					log(`${RED}💀${RESET} Queue destroyed`)
					syncImmediate()
				})
				stopPolling()
			}
			break
	}
})

// === Startup

void (async () => {
	await queue.restore()

	// Mark already-terminal entries so the watcher does not re-announce them
	for (const entry of queue.entries()) {
		if (isTerminalStatus(entry.status)) {
			loggedResults.add(entry.id)
		}
	}

	// Sync taskCounter to avoid ID collisions with restored entries
	let maxId = 0
	for (const entry of queue.entries()) {
		const match = /^(?:task|seq|bail|sig|fast|delay|prio|head|exp|w|nwd|nws)-(\d+)$/.exec(entry.id)
		const suffix = match?.[1]
		if (suffix !== undefined) {
			const num = parseInt(suffix, 10)
			if (num > maxId) maxId = num
		}
	}
	if (maxId > 0) taskCounter.set(maxId)

	const pending = queue
		.entries()
		.filter((e) => e.status === 'pending' || e.status === 'scheduled').length

	batcher.run(() => {
		syncImmediate()
		if (pending > 0) {
			log(`${GREEN}📂${RESET} Restored ${BOLD}${pending}${RESET} entries from previous session`)
		} else {
			log(
				`${DIM}Ready — press ${BOLD}a${RESET}${DIM} to add tasks, ${BOLD}b${RESET}${DIM} for batch, see commands below${RESET}`,
			)
		}
	})

	if (shouldPoll()) startPolling()
})()
