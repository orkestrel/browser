/**
 * BrowserPage
 *
 * Wraps a Playwright Page or Frame behind the BrowserPageInterface.
 * Pages and frames share the same API surface, so the consumer never
 * needs to learn two interfaces. Frame targets omit close() (no-op)
 * and throw on screenshot() since only Pages support it.
 */

import type {
	PlaywrightPageLike,
	BrowserPageInterface,
	BrowserNavigationOptions,
	BrowserActionOptions,
	BrowserScreenshotOptions,
	BrowserContentResult,
	BrowserScreenshotResult,
} from '../types.js'
import { BrowserError } from '../errors.js'
import { isPlaywrightPageLike } from '../helpers.js'

// === BrowserPage

export class BrowserPage implements BrowserPageInterface {
	#target: PlaywrightPageLike
	#isFrame: boolean

	constructor(target: unknown, isFrame: boolean) {
		if (!isPlaywrightPageLike(target)) {
			throw new BrowserError('Invalid page target: missing required Playwright methods')
		}
		this.#target = target
		this.#isFrame = isFrame
	}

	// === Property accessors

	get url(): string {
		return this.#target.url()
	}

	get closed(): boolean {
		const target = this.#target
		if (typeof target.isClosed === 'function') return target.isClosed()
		if (typeof target.isDetached === 'function') return target.isDetached()
		return false
	}

	// === Public API

	async title(): Promise<string> {
		return this.#target.title()
	}

	async navigate(url: string, options?: BrowserNavigationOptions): Promise<void> {
		const opts: Record<string, unknown> = {}
		if (options?.condition !== undefined) {
			opts['waitUntil'] = options.condition
		}
		if (options?.timeout !== undefined) {
			opts['timeout'] = options.timeout
		}
		await this.#target.goto(url, opts)
	}

	async content(): Promise<BrowserContentResult> {
		const [pageTitle, html] = await Promise.all([this.#target.title(), this.#target.content()])
		const text = await this.#target.evaluate('document.body ? document.body.innerText : ""')
		return {
			url: this.#target.url(),
			title: pageTitle,
			html,
			text: typeof text === 'string' ? text : '',
		}
	}

	async screenshot(options?: BrowserScreenshotOptions): Promise<BrowserScreenshotResult> {
		const target = this.#target
		if (target.screenshot === undefined) {
			throw new BrowserError('Screenshot is not available on frame targets')
		}
		const opts: Record<string, unknown> = {}
		if (options?.path !== undefined) opts['path'] = options.path
		if (options?.full !== undefined) opts['fullPage'] = options.full
		if (options?.type !== undefined) opts['type'] = options.type
		if (options?.quality !== undefined) opts['quality'] = options.quality
		const buffer = await target.screenshot(opts)
		return {
			bytes: new Uint8Array(buffer),
			path: options?.path ?? undefined,
		}
	}

	async click(selector: string, options?: BrowserActionOptions): Promise<void> {
		const opts: Record<string, unknown> = {}
		if (options?.timeout !== undefined) opts['timeout'] = options.timeout
		await this.#target.click(selector, opts)
	}

	async fill(selector: string, value: string, options?: BrowserActionOptions): Promise<void> {
		const opts: Record<string, unknown> = {}
		if (options?.timeout !== undefined) opts['timeout'] = options.timeout
		await this.#target.fill(selector, value, opts)
	}

	async select(
		selector: string,
		values: readonly string[],
		options?: BrowserActionOptions,
	): Promise<void> {
		const opts: Record<string, unknown> = {}
		if (options?.timeout !== undefined) opts['timeout'] = options.timeout
		await this.#target.selectOption(selector, values, opts)
	}

	async evaluate(expression: string): Promise<unknown> {
		return this.#target.evaluate(expression)
	}

	async wait(selector: string, options?: BrowserActionOptions): Promise<void> {
		const opts: Record<string, unknown> = {}
		if (options?.timeout !== undefined) opts['timeout'] = options.timeout
		await this.#target.waitForSelector(selector, opts)
	}

	frame(name: string): BrowserPageInterface | undefined {
		const target = this.#target
		if (target.frame === undefined) return undefined
		const child = target.frame(name)
		if (child === null) return undefined
		return new BrowserPage(child, true)
	}

	frames(): readonly BrowserPageInterface[] {
		const target = this.#target
		if (target.childFrames === undefined) return []
		const children = target.childFrames()
		const result: BrowserPageInterface[] = []
		for (const child of children) {
			result.push(new BrowserPage(child, true))
		}
		return result
	}

	async close(): Promise<void> {
		if (this.#isFrame) return
		if (typeof this.#target.close === 'function') {
			await this.#target.close()
		}
	}
}
